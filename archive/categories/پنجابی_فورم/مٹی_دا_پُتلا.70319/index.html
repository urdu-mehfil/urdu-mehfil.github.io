مٹی دا پُتلا

ساقی۔ — دسمبر 20، 2013 7:28 شام
مٹی دا پُتلا
مٹی میری فرض عبادت
مٹی نفل شبینہ
مٹی دے میں دیوے بالاں
مٹی نُور خزینہ
مٹی میرے علم دا منبہ
مٹی حرف لکیراں
مٹی تُوں میں سب کجھ سکھیا
مٹی مرشد پیراں
مٹی میرا زیور گہنا
مٹی کپڑے لتے
مٹی لفظ، کتاباں، قلماں
مٹی کاغذ گتے
مٹی دے میں سُفنے ویکھاں
مٹی نُوں تعبیراں
مٹی اُتے نظماں لکھاں
مٹی نُوں تصویراں
مٹی خواب ازل توں میرا
مٹی ابداں ماری
مٹی توں میں جند چھڑاواں
مٹی جان توں پیاری
مٹی دی میں کھیڈ بناواں
مٹی روندھ وی ماراں
مٹی نال میں بازی لاواں
مٹی جِتاں ہاراں
مٹی دے میں گنجل کڈاں
مٹی زلف سنواراں
مٹی اَکھ میں سرمہ پاواں
مٹی ہار سنگھاراں
مٹی توں میں صدقے جاواں
مٹی توں جند واراں
مٹی دے میں شگن مناواں
مٹی کھٹ کھلاراں
مٹی باہروں کجھ نہ دسدی
مٹی اندر خانے
مٹی جیہڑا آپے ہووے
مٹی نوں اوہ جانے
مٹی شوہ دریاواں ڈونگی
مٹی ٹھل طغیانی
مٹی دے وچ عمراں ڈُبیاں
مٹی اکھ دا پانی
مٹی پتنوں پار پچاوے
مٹی ریتو ڈَوبے
مٹی کنڈے، ٹوئے، ٹبے
مٹی کھائیاں کھوبے
مٹی وانگ سدائیاں ہسدی
مٹی اُڈ پُڈ جانی
مٹی رو رو ڈھائیں مارے
مٹی ذات نمانی
مٹی خوشبو، پُھل پھلواری
مٹی رات دی رانی
مٹی قصہ قاف پری دا
مٹی لوک کہانی
مٹی درد وچھوڑے پاؤندی
مٹی روگ فقیراں
مٹی دے نے مرزے رانجھے
مٹی صاحباں ہیراں
مٹی دے مہینوال تے سوہنیاں
مٹی سسی پنوں
مٹی دے فرہاد تے شیریں
مٹی لیلی مجنوں
مٹی یوسف، مٹی زلیخاں
مٹی سوتر اٹی
مٹی مصر بزاراں وکدی
مٹی مول نہ ہٹی
مٹی وچ درباراں نچدی
مٹی پائے دھمالاں
مٹی جنگل بیلے پھر دی
مٹی یار وصالاں
مٹی شاہ لطیف دی وائی
مٹی خواجہ، باہو
مٹی شاہ حسین دی کافی
مٹی آلا، یاہُو
مٹی گنج شکر تے داتا
مٹی بلھے شاہ جی
مٹی گامے شاہ دی کربل
مٹی رابعہ بصری
مٹی شاہ عنائت سوہنا
مٹی مادھو لال
مٹی وارث، میاں محمد
مٹی شاہ جمال
مٹی تن من اندر چمکے
مٹی لشکاں مارے
مٹی دور ہنیرے کر دی
مٹی چانن تارے
مٹی چاندی، مٹی سونا
مٹی ہیرے موتی
مٹی پُتر، ویر اکلا
مٹی دھی اکلوتی
مٹی جائے نماز مسیتاں
مٹی مریم، سیتا
مٹی گوتم، مٹی یشودھا
مٹی گوپی، رادھا
مٹی دے نیں آدم حوا
مٹی عشق خدا دا
مٹی دا اعجاز وکھاوے
مٹی دا اک پتلا!
